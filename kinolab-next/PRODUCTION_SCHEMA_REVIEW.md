## Recommended first reviewed DDL (do NOT run yet)

This is a **catalog-only candidate**, not a complete production migration and not a replacement for locally created Supabase CLI migration with tests. Review against up-to-date Supabase RLS guidance and the current staging schema before execution.

```sql
-- REVIEW ONLY: apply to the approved production project only after approval.
BEGIN;

CREATE SCHEMA IF NOT EXISTS cinewave_private;
REVOKE ALL ON SCHEMA cinewave_private FROM PUBLIC, anon, authenticated;

CREATE TABLE IF NOT EXISTS cinewave_private.admin_users (
  user_id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  assigned_at timestamptz NOT NULL DEFAULT now()
);
ALTER TABLE cinewave_private.admin_users ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON TABLE cinewave_private.admin_users FROM PUBLIC, anon, authenticated;

CREATE OR REPLACE FUNCTION public.cinewave_is_admin()
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = ''
AS $$
  SELECT auth.uid() IS NOT NULL
     AND EXISTS (
       SELECT 1 FROM cinewave_private.admin_users AS a
       WHERE a.user_id = auth.uid()
     );
$$;
REVOKE ALL ON FUNCTION public.cinewave_is_admin() FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION public.cinewave_is_admin() TO authenticated;

CREATE TABLE IF NOT EXISTS public.kinolab_external_catalog (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title_en text NOT NULL,
  title_ru text,
  title_hy text,
  overview_en text,
  overview_ru text,
  overview_hy text,
  release_date date,
  genre_names text[] NOT NULL DEFAULT '{}',
  poster_path text,
  external_watch_url text NOT NULL,
  is_published boolean NOT NULL DEFAULT false,
  terms_confirmed boolean NOT NULL DEFAULT false,
  created_at timestamptz NOT NULL DEFAULT now(),
  trailer_url text,
  media_type text NOT NULL DEFAULT 'movie',
  episode_links jsonb NOT NULL DEFAULT '[]'::jsonb,
  CONSTRAINT external_catalog_unique_link UNIQUE (external_watch_url),
  CONSTRAINT external_catalog_publication CHECK ((NOT is_published) OR terms_confirmed),
  CONSTRAINT kinolab_external_catalog_title_en_check CHECK (length(btrim(title_en)) BETWEEN 1 AND 250),
  CONSTRAINT kinolab_external_catalog_https_link CHECK (
    length(external_watch_url) <= 2048
    AND external_watch_url ~ '^https://[^[:space:]@/?#]+'
    AND external_watch_url !~ '[[:cntrl:][:space:]]'
  )
);

ALTER TABLE public.kinolab_external_catalog ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON TABLE public.kinolab_external_catalog FROM PUBLIC, anon, authenticated;
GRANT SELECT ON TABLE public.kinolab_external_catalog TO anon;
GRANT SELECT, INSERT, UPDATE, DELETE ON TABLE public.kinolab_external_catalog TO authenticated;

CREATE POLICY kinolab_public_read ON public.kinolab_external_catalog
 FOR SELECT TO anon, authenticated USING (is_published);
CREATE POLICY kinolab_admin_read ON public.kinolab_external_catalog
 FOR SELECT TO authenticated USING ((SELECT public.cinewave_is_admin()));
CREATE POLICY kinolab_admin_insert ON public.kinolab_external_catalog
 FOR INSERT TO authenticated WITH CHECK ((SELECT public.cinewave_is_admin()));
CREATE POLICY kinolab_admin_update ON public.kinolab_external_catalog
 FOR UPDATE TO authenticated
 USING ((SELECT public.cinewave_is_admin()))
 WITH CHECK ((SELECT public.cinewave_is_admin()));
CREATE POLICY kinolab_admin_delete ON public.kinolab_external_catalog
 FOR DELETE TO authenticated USING ((SELECT public.cinewave_is_admin()));

COMMIT;
```
